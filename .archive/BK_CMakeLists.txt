        headers/hex.hpp
        headers/state.hpp
        headers/column.hpp
        headers/row_labels.hpp
        src/board.cpp
        headers/board.hpp
        src/gameManager.cpp
        headers/gameManager.hpp
)

        Qt::Core
        Qt::Gui
        Qt::Widgets
